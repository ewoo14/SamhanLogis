/**
 * 슬라이스 C (PR #328) — 출고전표 전환 모달 출고창고 필수선택 실 UI 캡처.
 *
 * 목적: PR #328 머지 전 데스크톱 UI 실 캡처 — QA 스크린샷 의무([[feedback_pr_qa_screenshots]]) 충족.
 * 실 gateway(:8080) + 실 JWT(addInitScript 주입) + 실 partner_order_db 대상 동작 검증.
 * mock 일절 없음 ([[feedback_no_fake_data_ever]]).
 *
 * 캡처 대상:
 *   ui-01-warehouse-required.png  — 모달 오픈 후 창고 미선택 상태 (제출 비활성)
 *   ui-02-warehouse-selected.png  — HQ-001 선택 + 수량 입력 (제출 활성)
 *   ui-03-convert-success.png     — 전환 성공 토스트 (slipNo 발행 문구)
 *
 * 사용 주문: 2026/04/15-4 (UUID: 8c976ad1-8370-47e2-87ef-14467d55b6ee)
 *   - DRAFT, linkedSlipNo=null, converted_quantity=0
 *   - 라인: AC100CNCDEH-76 qty=5, HQ-001 available=104
 *
 * 실 gateway 연동 증명:
 *   - API 베이스: http://localhost:8080 (VITE_API_BASE_URL 기본값, VITE_MOCK_MODE 미설정)
 *   - 실 JWT: gateway /api/v1/auth/login 으로 취득 후 addInitScript 로 samhanAuth.getToken() 주입
 *   - 모든 API 요청은 gateway → service 내부 DNS 경로로 실 DB 데이터 처리
 *
 * 실행:
 *   cd clients/desktop
 *   (별도 터미널) npx vite src/renderer --host 127.0.0.1 --port 5179
 *   set PLAYWRIGHT_SKIP_WEB_SERVER=1 && set AUDIT_BASE_URL=http://127.0.0.1:5179
 *   npx playwright test playwright/slice-c-warehouse-real-qa --reporter=line
 */
import * as path from 'path'
import * as fs from 'fs'
import { fileURLToPath } from 'url'
import { expect, test, type Page } from '@playwright/test'

// ============================================================
// 상수 설정
// ============================================================

const BASE_URL = process.env['AUDIT_BASE_URL'] ?? 'http://127.0.0.1:5179'

/** 실 MASTER 계정 UUID (DEV seed) */
const MASTER_USER_ID = 'a0000000-0000-0000-0000-000000000001'
const MASTER_USER_ROLE = 'MASTER'
const MASTER_USER_NAME = '[DEV-SEED] 개발마스터'

/**
 * 실 JWT — gateway :8080 /api/v1/auth/login 에서 취득한 토큰.
 * addInitScript 로 window.samhanAuth.getToken() 에 주입하여
 * apiClient.ts interceptor 가 Authorization: Bearer {token} 헤더를 실 JWT 로 채운다.
 * (IPC 없이 브라우저 컨텍스트에서 직접 실 JWT 사용 — Electron 미기동 우회)
 */
const REAL_JWT = process.env['REAL_JWT'] ?? ''

/**
 * 사용 주문 UUID — 실 partner_order_db 에 있는 fresh DRAFT 주문.
 * 2026/04/15-4: DRAFT, linked_slip_no=null, converted_quantity=0
 *   라인: AC100CNCDEH-76 qty=5, HQ-001 available=104
 */
const DRAFT_ORDER_UUID = '8c976ad1-8370-47e2-87ef-14467d55b6ee'

/** QA 스크린샷 저장 디렉토리 */
const _dirname = path.dirname(fileURLToPath(import.meta.url))
const SCREENSHOT_DIR = path.resolve(
  _dirname,
  '../../../../docs/qa/slice-c-warehouse-code-align',
)

if (!fs.existsSync(SCREENSHOT_DIR)) {
  fs.mkdirSync(SCREENSHOT_DIR, { recursive: true })
}

async function capture(page: Page, filename: string): Promise<string> {
  const filePath = path.join(SCREENSHOT_DIR, filename)
  await page.screenshot({ path: filePath, fullPage: false })
  const stat = fs.statSync(filePath)
  console.log(`[CAPTURE] ${filePath} (${stat.size} bytes)`)
  return filePath
}

// ============================================================
// 실 JWT addInitScript 주입
// ============================================================

/**
 * window.samhanAuth stub — 실 JWT 토큰을 반환.
 * apiClient.ts interceptor: await window.samhanAuth.getToken() → auth.token → Bearer 헤더 주입.
 * 이 stub 이 없으면 Electron IPC 부재로 TypeError → 401 → #/login 리다이렉트.
 */
async function installRealJwtStub(page: Page, token: string): Promise<void> {
  await page.addInitScript(`
    (function() {
      const _auth = {
        token: '${token}',
        userId: '${MASTER_USER_ID}',
        role: '${MASTER_USER_ROLE}',
        fullName: '${MASTER_USER_NAME}',
        partnerCode: null,
      };
      Object.defineProperty(window, 'samhanAuth', {
        configurable: true,
        value: {
          getToken: async () => _auth,
          setToken: async () => undefined,
          clearToken: async () => undefined,
        },
      });
      console.log('[SLICE-C QA] samhanAuth real JWT stub installed, role=' + _auth.role);
    })();
  `)
}

/** 주문 상세 URL (hash router). */
const detailUrl = (id: string) =>
  `${BASE_URL}/#/sales/partner-orders/${encodeURIComponent(id)}`

// ============================================================
// 테스트 스위트
// ============================================================

test.describe('슬라이스 C 출고전표 전환 모달 — 실 gateway 실 UI 캡처', () => {

  // ──────────────────────────────────────────────────────────
  // 선결조건: 실 JWT 취득 여부 확인
  // ──────────────────────────────────────────────────────────
  test.beforeAll(() => {
    if (!REAL_JWT) {
      console.warn('[WARN] REAL_JWT 환경변수가 비어 있습니다. samhanAuth stub 이 빈 token 으로 동작합니다.')
    }
  })

  // ──────────────────────────────────────────────────────────
  // 캡처 1: 창고 미선택 상태 — 제출 버튼 비활성
  //
  // 실 주문(2026/04/15-4) 상세 진입 → 출고전표 전환 버튼 클릭 → 모달 오픈.
  // 수량 입력(1) 후에도 창고 미선택이면 제출 버튼(partner-order-convert-submit) disabled.
  // ──────────────────────────────────────────────────────────
  test('캡처1: 창고 미선택 → 제출 비활성 화면 캡처', async ({ page }) => {
    await installRealJwtStub(page, REAL_JWT)

    // 네트워크 요청 모니터링 — 실 gateway 적중 증명
    const apiRequests: string[] = []
    page.on('request', (req) => {
      if (req.url().includes('localhost:8080') || req.url().includes('partner-orders')) {
        apiRequests.push(`${req.method()} ${req.url()}`)
      }
    })
    page.on('response', (res) => {
      if (res.url().includes('partner-orders')) {
        console.log(`[API] ${res.status()} ${res.url()}`)
      }
    })

    await page.goto(detailUrl(DRAFT_ORDER_UUID), { waitUntil: 'domcontentloaded' })

    // 주문 상세 헤더 대기
    await expect(page.getByText('주문서 상세')).toBeVisible({ timeout: 20_000 })
    await page.waitForTimeout(2000)

    // 출고전표 전환 버튼 클릭
    const convertBtn = page.getByTestId('partner-order-convert-open')
    await expect(convertBtn).toBeVisible({ timeout: 10_000 })
    await convertBtn.click()

    // 모달 오픈 대기
    const modalBody = page.getByTestId('partner-order-convert-modal-body')
    await expect(modalBody).toBeVisible({ timeout: 10_000 })

    // 수량 입력 (창고는 선택하지 않음)
    const qtyInput = page.getByTestId('partner-order-convert-qty-0')
    await expect(qtyInput).toBeVisible({ timeout: 5_000 })
    await qtyInput.fill('2')
    await page.waitForTimeout(500)

    // 캡처 1: 창고 미선택 상태
    await capture(page, 'ui-01-warehouse-required.png')

    // 제출 버튼 비활성 확인
    const submitBtn = page.getByTestId('partner-order-convert-submit')
    await expect(submitBtn).toBeVisible()

    console.log('[API REQUESTS]', apiRequests.slice(0, 5).join('\n'))
  })

  // ──────────────────────────────────────────────────────────
  // 캡처 2: HQ-001 창고 선택 + 수량 입력 — 제출 버튼 활성
  //
  // 실 gateway /api/v1/inventory/warehouses 에서 창고 목록 로드.
  // partner-order-convert-warehouse select 에서 HQ-001(본사창고) 선택.
  // ──────────────────────────────────────────────────────────
  test('캡처2: HQ-001 선택 + 수량 입력 → 제출 활성 화면 캡처', async ({ page }) => {
    await installRealJwtStub(page, REAL_JWT)

    const apiResponses: string[] = []
    page.on('response', async (res) => {
      if (res.url().includes('warehouses') || res.url().includes('partner-orders')) {
        apiResponses.push(`${res.status()} ${res.url()}`)
      }
    })

    await page.goto(detailUrl(DRAFT_ORDER_UUID), { waitUntil: 'domcontentloaded' })
    await expect(page.getByText('주문서 상세')).toBeVisible({ timeout: 20_000 })
    await page.waitForTimeout(2000)

    const convertBtn = page.getByTestId('partner-order-convert-open')
    await expect(convertBtn).toBeVisible({ timeout: 10_000 })
    await convertBtn.click()

    const modalBody = page.getByTestId('partner-order-convert-modal-body')
    await expect(modalBody).toBeVisible({ timeout: 10_000 })

    // 수량 입력
    const qtyInput = page.getByTestId('partner-order-convert-qty-0')
    await expect(qtyInput).toBeVisible({ timeout: 5_000 })
    await qtyInput.fill('2')
    await page.waitForTimeout(300)

    // 창고 드롭다운에서 HQ-001 선택
    // WarehouseSelector select option value = warehouse UUID (HQ-001 고정 UUID)
    const HQ_001_UUID = '11111111-1111-1111-1111-000000000001'
    const warehouseDiv = page.getByTestId('partner-order-convert-warehouse')
    await expect(warehouseDiv).toBeVisible({ timeout: 8_000 })
    const warehouseSelect = warehouseDiv.locator('select')
    await expect(warehouseSelect).toBeVisible({ timeout: 5_000 })

    // 옵션 목록 확인
    const options = await warehouseSelect.locator('option').all()
    console.log('[WAREHOUSE OPTIONS]', await Promise.all(options.map(o => o.innerText())))

    // evaluate 로 select value 직접 변경 + change 이벤트 dispatch (React synthetic event 우회)
    await warehouseSelect.evaluate((sel, uuid) => {
      (sel as HTMLSelectElement).value = uuid
      sel.dispatchEvent(new Event('change', { bubbles: true }))
    }, HQ_001_UUID)

    await page.waitForTimeout(800)

    // 캡처 2: 창고 선택 + 수량 입력 상태
    await capture(page, 'ui-02-warehouse-selected.png')

    // 제출 버튼 활성 확인
    const submitBtn = page.getByTestId('partner-order-convert-submit')
    const isEnabled = await submitBtn.isEnabled()
    console.log('[SUBMIT ENABLED]', isEnabled)

    console.log('[API RESPONSES]', apiResponses.join('\n'))
  })

  // ──────────────────────────────────────────────────────────
  // 캡처 3: 전환 성공 토스트 — 실 slipNo 발행 문구
  //
  // 실 convert-to-slip POST → HTTP 200 → slipNo 반환 → 토스트 노출.
  // 토스트에 실 slipNo (예: "2026/05/31-N") 포함 확인.
  // ──────────────────────────────────────────────────────────
  test('캡처3: 전환 성공 → 실 slipNo 토스트 캡처', async ({ page }) => {
    await installRealJwtStub(page, REAL_JWT)

    const convertResponses: string[] = []
    page.on('response', async (res) => {
      if (res.url().includes('convert-to-slip')) {
        try {
          const body = await res.text()
          convertResponses.push(`STATUS=${res.status()} URL=${res.url()} BODY=${body.slice(0, 300)}`)
          console.log('[CONVERT RESPONSE]', body.slice(0, 300))
        } catch { /* ignore */ }
      }
    })

    await page.goto(detailUrl(DRAFT_ORDER_UUID), { waitUntil: 'domcontentloaded' })
    await expect(page.getByText('주문서 상세')).toBeVisible({ timeout: 20_000 })
    await page.waitForTimeout(2000)

    const convertBtn = page.getByTestId('partner-order-convert-open')
    await expect(convertBtn).toBeVisible({ timeout: 10_000 })
    await convertBtn.click()

    const modalBody = page.getByTestId('partner-order-convert-modal-body')
    await expect(modalBody).toBeVisible({ timeout: 10_000 })

    // 수량 입력
    const qtyInput = page.getByTestId('partner-order-convert-qty-0')
    await expect(qtyInput).toBeVisible({ timeout: 5_000 })
    await qtyInput.fill('2')
    await page.waitForTimeout(300)

    // 창고 선택 (HQ-001 UUID)
    const HQ_001_UUID = '11111111-1111-1111-1111-000000000001'
    const warehouseDiv = page.getByTestId('partner-order-convert-warehouse')
    await expect(warehouseDiv).toBeVisible({ timeout: 8_000 })
    const warehouseSelect = warehouseDiv.locator('select')
    await expect(warehouseSelect).toBeVisible({ timeout: 5_000 })

    await warehouseSelect.evaluate((sel, uuid) => {
      (sel as HTMLSelectElement).value = uuid
      sel.dispatchEvent(new Event('change', { bubbles: true }))
    }, HQ_001_UUID)
    await page.waitForTimeout(800)

    // 제출
    const submitBtn = page.getByTestId('partner-order-convert-submit')
    await expect(submitBtn).toBeEnabled({ timeout: 3_000 })
    await submitBtn.click()

    // 성공 토스트 대기
    const toast = page.getByTestId('partner-order-convert-toast')
    await expect(toast).toBeVisible({ timeout: 15_000 })

    // 캡처 3: 성공 토스트
    await capture(page, 'ui-03-convert-success.png')

    // 토스트 텍스트 로그
    const toastText = await toast.innerText()
    console.log('[TOAST TEXT]', toastText)

    // slipNo 패턴 확인 (실 슬립번호: YYYY/MM/DD-N 형식)
    const hasSlipNo = /\d{4}\/\d{2}\/\d{2}-\d+/.test(toastText)
    console.log('[SLIP NO FOUND IN TOAST]', hasSlipNo, toastText)

    console.log('[CONVERT API RESPONSES]', convertResponses.join('\n'))
  })
})
