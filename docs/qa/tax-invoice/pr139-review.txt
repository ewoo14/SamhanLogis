P04ValidationIT.cancelChangesStatusToCancelledWithReverseJournalId() 라인 265:

### 4. 도메인 정합성 SQL 검증 쿼리 (실 DB 적용 시)

```sql
-- VAT 10% invariant
SELECT id, tax_invoice_no, supply_amount, vat_amount, total_amount
FROM tax_invoices
WHERE is_deleted = FALSE
  AND (ABS(vat_amount - ROUND(supply_amount * 0.10, 2)) > 1
    OR ABS(total_amount - (supply_amount + vat_amount)) > 0.01);
-- 결과 0건이어야 함

-- 라인 합계 vs 헤더 합계
SELECT ti.id, ti.tax_invoice_no, ti.supply_amount, SUM(l.supply_amount) AS line_sum
FROM tax_invoices ti
JOIN tax_invoice_lines l ON l.tax_invoice_id = ti.id AND l.is_deleted = FALSE
WHERE ti.is_deleted = FALSE
GROUP BY ti.id, ti.tax_invoice_no, ti.supply_amount
HAVING ABS(ti.supply_amount - SUM(l.supply_amount)) > 0.01;
-- 결과 0건이어야 함

-- invoiceNo unique (ISSUED 상태)
SELECT tax_invoice_no, COUNT(*) FROM tax_invoices
WHERE is_deleted = FALSE AND status = 'ISSUED' AND tax_invoice_no IS NOT NULL
GROUP BY tax_invoice_no HAVING COUNT(*) > 1;
-- 결과 0건이어야 함

-- cancel_reason 5자 미만 (CANCELLED 상태)
SELECT id, tax_invoice_no, cancel_reason FROM tax_invoices
WHERE is_deleted = FALSE AND status = 'CANCELLED'
  AND (cancel_reason IS NULL OR LENGTH(TRIM(cancel_reason)) < 5);
-- 결과 0건이어야 함

-- 사업자등록번호 형식 위반
SELECT id, tax_invoice_no, partner_business_no FROM tax_invoices
WHERE is_deleted = FALSE AND partner_business_no IS NOT NULL
  AND partner_business_no !~ '^\d{3}-\d{2}-\d{5}$';
-- 결과 0건이어야 함
```
