/**
 * AC-2 (PR #332) — 품목 자동완성(ProductAutocomplete) 실 gateway 실 UI 캡처.
 *
 * 목적: PR #332 머지 전 데스크톱 UI 실 캡처 — QA 스크린샷 의무([[feedback_pr_qa_screenshots]]) 충족.
 * 실 gateway(:8080) + 실 JWT(addInitScript 주입) + 실 product-service 데이터 대상 동작 검증.
 * mock 일절 없음 ([[feedback_no_fake_data_ever]]).
 *
 * 캡처 대상:
 *   ui-01-product-search.png  — "AR05" 타이핑 후 실 후보 드롭다운 노출
 *   ui-02-product-selected.png — 후보 선택 완료 + 단가 자동 채움
 *
 * 실 품목 시드 (product-service, 100건):
 *   - 검색어 "AR05" → AR05TXEAAWKNEU-01 등 3건 (삼성 윈드프리 5평형, sellingPrice: 600000)
 *   - GET /api/products?q=AR05&size=20 → HTTP 200, totalElements=3
 *
 * 실 gateway 연동 증명:
 *   - API 베이스: http://localhost:8080 (VITE_MOCK_MODE 미설정)
 *   - 실 JWT: gateway /api/v1/auth/login dev_master 계정 취득 후 addInitScript 주입
 *   - GET /api/products?q=AR05 → AR05TXEAAWKNEU-01 후보 실 데이터
 *
 * 실행:
 *   cd clients/desktop
 *   (별도 터미널) npx vite src/renderer --host 127.0.0.1 --port 5179
 *   set PLAYWRIGHT_SKIP_WEB_SERVER=1 && set AUDIT_BASE_URL=http://127.0.0.1:5179
 *   set REAL_JWT=<MASTER_TOKEN>
 *   npx playwright test playwright/ac-2-product-autocomplete/ac2-real-qa-driver --reporter=line
 */
import * as path from 'path'
import * as fs from 'fs'
import { fileURLToPath } from 'url'
import { expect, test, type Page } from '@playwright/test'

// ============================================================
// 상수 설정
// ============================================================

const BASE_URL = process.env['AUDIT_BASE_URL'] ?? 'http://127.0.0.1:5179'

const MASTER_USER_ID = 'a0000000-0000-0000-0000-000000000001'
const MASTER_USER_ROLE = 'MASTER'
const MASTER_USER_NAME = '[DEV-SEED] 개발마스터'

/**
 * 실 JWT — gateway :8080 /api/v1/auth/login 에서 취득한 토큰.
 * addInitScript 로 window.samhanAuth.getToken() 에 주입.
 */
const REAL_JWT = process.env['REAL_JWT'] ?? ''

const _dirname = path.dirname(fileURLToPath(import.meta.url))
const SCREENSHOT_DIR = path.resolve(
  _dirname,
  '../../../../docs/qa/ac2-product-autocomplete',
)

if (!fs.existsSync(SCREENSHOT_DIR)) {
  fs.mkdirSync(SCREENSHOT_DIR, { recursive: true })
}

async function capture(page: Page, filename: string): Promise<string> {
  const filePath = path.join(SCREENSHOT_DIR, filename)
  await page.screenshot({ path: filePath, fullPage: false })
  const stat = fs.statSync(filePath)
  console.log(`[CAPTURE] ${filePath} (${stat.size} bytes)`)
  return filePath
}

// ============================================================
// 실 JWT addInitScript 주입
// ============================================================

async function installRealJwtStub(page: Page, token: string): Promise<void> {
  await page.addInitScript(`
    (function() {
      const _auth = {
        token: '${token}',
        userId: '${MASTER_USER_ID}',
        role: '${MASTER_USER_ROLE}',
        fullName: '${MASTER_USER_NAME}',
        partnerCode: null,
      };
      Object.defineProperty(window, 'samhanAuth', {
        configurable: true,
        value: {
          getToken: async () => _auth,
          setToken: async () => undefined,
          clearToken: async () => undefined,
        },
      });
      console.log('[AC-2 QA] samhanAuth real JWT stub installed, role=' + _auth.role);
    })();
  `)
}

const slipNewUrl = () => `${BASE_URL}/#/sales/new`

// ============================================================
// 테스트 스위트
// ============================================================

test.describe('AC-2 품목 자동완성 — 실 gateway 실 UI 캡처', () => {

  test.beforeAll(() => {
    if (!REAL_JWT) {
      throw new Error('[BLOCKED] REAL_JWT 환경변수가 없음. 실 JWT 없이 실 gateway 캡처 불가.')
    }
  })

  // ──────────────────────────────────────────────────────────
  // 캡처 1: "AR05" 타이핑 → 실 후보 드롭다운 노출
  //
  // 실 gateway GET /api/products?q=AR05 에서 품목 목록 로드.
  // ProductAutocomplete input(role=combobox) 에 "AR05" 타이핑 → listbox[role=listbox] 노출.
  // AR05TXEAAWKNEU-01 후보가 candidates 에 포함됨을 실 화면으로 증명.
  // ──────────────────────────────────────────────────────────
  test('캡처1: "AR05" 타이핑 → 실 품목 후보 드롭다운 노출', async ({ page }) => {
    await installRealJwtStub(page, REAL_JWT)

    // 실 gateway 네트워크 요청 모니터링
    const productApiHits: string[] = []
    page.on('response', async (res) => {
      if (res.url().includes('/api/products')) {
        productApiHits.push(`${res.status()} GET ${res.url()}`)
        console.log(`[API HIT] products: ${res.status()} ${res.url()}`)
      }
    })

    await page.goto(slipNewUrl(), { waitUntil: 'domcontentloaded' })

    // 전표 작성 화면 로드 완료 대기 — "+ 라인 추가" 버튼 표시
    await expect(page.getByRole('button', { name: '+ 라인 추가' })).toBeVisible({ timeout: 20_000 })
    await page.waitForTimeout(1500)

    // ProductAutocomplete input — 라인 1 품목 (role=combobox, ariaLabel="라인 1 품목")
    const productInput = page.getByRole('combobox', { name: /라인 1 품목/ })
    await expect(productInput).toBeVisible({ timeout: 8_000 })

    // "AR05" 타이핑
    await productInput.click()
    await productInput.fill('AR05')

    // listbox 노출 대기 (aria-expanded=true + 실 API 응답 완료)
    await expect(productInput).toHaveAttribute('aria-expanded', 'true', { timeout: 8_000 })

    const listbox = page.getByRole('listbox', { name: '품목 목록' })
    await expect(listbox).toBeVisible({ timeout: 8_000 })

    // AR05TXEAAWKNEU-01 후보 확인
    const firstOption = listbox.getByRole('option').first()
    await expect(firstOption).toBeVisible({ timeout: 5_000 })
    const optionText = await firstOption.innerText()
    console.log('[AUTOCOMPLETE FIRST OPTION]', optionText)

    // 캡처 1: 후보 드롭다운 노출 상태
    await capture(page, 'ui-01-product-search.png')

    // 모든 후보 로깅
    const allOptions = await listbox.getByRole('option').all()
    const allTexts = await Promise.all(allOptions.map((o) => o.innerText()))
    console.log('[ALL AUTOCOMPLETE OPTIONS]', allTexts)

    // AR05 포함 증명
    const hasAr05 = allTexts.some((t) => t.includes('AR05'))
    console.log('[AR05 IN CANDIDATES]', hasAr05)
    expect(hasAr05).toBe(true)

    console.log('[PRODUCT API HITS]', productApiHits)
    expect(productApiHits.length).toBeGreaterThan(0)
  })

  // ──────────────────────────────────────────────────────────
  // 캡처 2: 후보 클릭 선택 → modelName 입력란 반영 + 단가 자동 채움
  //
  // listbox 첫 번째 후보 클릭 → input value = "AR05TXEAAWKNEU-01"
  // 단가 입력란에 600000 채워짐 확인 (sellingPrice).
  // ──────────────────────────────────────────────────────────
  test('캡처2: 후보 클릭 선택 → modelName 표시 + 단가 자동 채움', async ({ page }) => {
    await installRealJwtStub(page, REAL_JWT)

    const apiResponses: string[] = []
    page.on('response', async (res) => {
      if (res.url().includes('/api/products')) {
        apiResponses.push(`${res.status()} ${res.url()}`)
      }
    })

    await page.goto(slipNewUrl(), { waitUntil: 'domcontentloaded' })
    await expect(page.getByRole('button', { name: '+ 라인 추가' })).toBeVisible({ timeout: 20_000 })
    await page.waitForTimeout(1500)

    // ProductAutocomplete input — 라인 1 품목
    const productInput = page.getByRole('combobox', { name: /라인 1 품목/ })
    await expect(productInput).toBeVisible({ timeout: 8_000 })

    // "AR05" 타이핑 → 후보 노출
    await productInput.click()
    await productInput.fill('AR05')
    await expect(productInput).toHaveAttribute('aria-expanded', 'true', { timeout: 8_000 })

    const listbox = page.getByRole('listbox', { name: '품목 목록' })
    await expect(listbox).toBeVisible({ timeout: 8_000 })

    // 첫 번째 후보 클릭 (AR05TXEAAWKNEU-01)
    const firstOption = listbox.getByRole('option').first()
    await expect(firstOption).toBeVisible({ timeout: 5_000 })
    const optionText = await firstOption.innerText()
    console.log('[CLICKING OPTION]', optionText)
    await firstOption.click()

    // 선택 후 input value = AR05TXEAAWKNEU-01 확인
    await expect(productInput).toHaveAttribute('aria-expanded', 'false', { timeout: 5_000 })
    const inputValue = await productInput.inputValue()
    console.log('[SELECTED INPUT VALUE]', inputValue)
    expect(inputValue).toMatch(/AR05/)

    // listbox 닫힘 확인
    await expect(listbox).not.toBeVisible({ timeout: 3_000 })

    // 단가 자동 채움 확인 — "라인 1 단가" textbox
    await page.waitForTimeout(300)
    const priceInput = page.getByRole('textbox', { name: '라인 1 단가' })
    await expect(priceInput).toBeVisible({ timeout: 5_000 })
    const priceVal = await priceInput.inputValue()
    const priceNum = Number(priceVal.replace(/,/g, ''))
    console.log('[UNIT PRICE VALUE]', priceVal, '→', priceNum)
    expect(priceNum).toBeGreaterThan(0)

    // 캡처 2: 선택 완료 + 단가 채움
    await capture(page, 'ui-02-product-selected.png')

    console.log('[API RESPONSES]', apiResponses.join('\n'))
  })

})
