/**
 * AC-1 (PR #331) — 창고 자동완성(WarehouseAutocomplete) 실 gateway 실 UI 캡처.
 *
 * 목적: PR #331 머지 전 데스크톱 UI 실 캡처 — QA 스크린샷 의무([[feedback_pr_qa_screenshots]]) 충족.
 * 실 gateway(:8080) + 실 JWT(addInitScript 주입) + 실 partner_order_db 대상 동작 검증.
 * mock 일절 없음 ([[feedback_no_fake_data_ever]]).
 *
 * 캡처 대상:
 *   ui-01-autocomplete-suggestions.png  — "본사" 타이핑 후 HQ-001 후보 드롭다운 노출
 *   ui-02-selected.png                  — HQ-001 선택 완료 + 수량 입력 + 제출 버튼 활성
 *   ui-03-convert-success.png           — 전환 성공 토스트 (실 slipNo 발행)
 *
 * 사용 주문: 2026/04/15-4 (UUID: 8c976ad1-8370-47e2-87ef-14467d55b6ee)
 *   - DRAFT, linkedSlipNo=null
 *   - 라인: AC100CNCDEH-76, HQ-001 available>0
 *
 * 실 gateway 연동 증명:
 *   - API 베이스: http://localhost:8080 (VITE_MOCK_MODE 미설정)
 *   - 실 JWT: gateway /api/v1/auth/login 으로 취득 후 addInitScript 로 주입
 *   - GET /api/v1/inventory/warehouses → HQ-001 후보 실 데이터
 *   - POST /api/v1/partner-orders/{id}/convert-to-slip → 실 slipNo
 *
 * 실행:
 *   cd clients/desktop
 *   (별도 터미널) npx vite src/renderer --host 127.0.0.1 --port 5179
 *   set PLAYWRIGHT_SKIP_WEB_SERVER=1 && set AUDIT_BASE_URL=http://127.0.0.1:5179
 *   set REAL_JWT=<MASTER_TOKEN>
 *   npx playwright test playwright/ac1-warehouse-autocomplete-real-qa --reporter=line
 */
import * as path from 'path'
import * as fs from 'fs'
import { fileURLToPath } from 'url'
import { expect, test, type Page } from '@playwright/test'

// ============================================================
// 상수 설정
// ============================================================

const BASE_URL = process.env['AUDIT_BASE_URL'] ?? 'http://127.0.0.1:5179'

const MASTER_USER_ID = 'a0000000-0000-0000-0000-000000000001'
const MASTER_USER_ROLE = 'MASTER'
const MASTER_USER_NAME = '[DEV-SEED] 개발마스터'

/**
 * 실 JWT — gateway :8080 /api/v1/auth/login 에서 취득한 토큰.
 * addInitScript 로 window.samhanAuth.getToken() 에 주입.
 */
const REAL_JWT = process.env['REAL_JWT'] ?? ''

/**
 * 사용 주문 UUID — 실 partner_order_db 의 DRAFT 주문.
 * 2026/04/15-4: DRAFT, linkedSlipNo=null, 라인: AC100CNCDEH-76
 */
const DRAFT_ORDER_UUID = '8c976ad1-8370-47e2-87ef-14467d55b6ee'

const _dirname = path.dirname(fileURLToPath(import.meta.url))
const SCREENSHOT_DIR = path.resolve(
  _dirname,
  '../../../../docs/qa/ac1-warehouse-autocomplete',
)

if (!fs.existsSync(SCREENSHOT_DIR)) {
  fs.mkdirSync(SCREENSHOT_DIR, { recursive: true })
}

async function capture(page: Page, filename: string): Promise<string> {
  const filePath = path.join(SCREENSHOT_DIR, filename)
  await page.screenshot({ path: filePath, fullPage: false })
  const stat = fs.statSync(filePath)
  console.log(`[CAPTURE] ${filePath} (${stat.size} bytes)`)
  return filePath
}

// ============================================================
// 실 JWT addInitScript 주입
// ============================================================

async function installRealJwtStub(page: Page, token: string): Promise<void> {
  await page.addInitScript(`
    (function() {
      const _auth = {
        token: '${token}',
        userId: '${MASTER_USER_ID}',
        role: '${MASTER_USER_ROLE}',
        fullName: '${MASTER_USER_NAME}',
        partnerCode: null,
      };
      Object.defineProperty(window, 'samhanAuth', {
        configurable: true,
        value: {
          getToken: async () => _auth,
          setToken: async () => undefined,
          clearToken: async () => undefined,
        },
      });
      console.log('[AC-1 QA] samhanAuth real JWT stub installed, role=' + _auth.role);
    })();
  `)
}

const detailUrl = (id: string) =>
  `${BASE_URL}/#/sales/partner-orders/${encodeURIComponent(id)}`

// ============================================================
// 테스트 스위트
// ============================================================

test.describe('AC-1 창고 자동완성 — 실 gateway 실 UI 캡처', () => {

  test.beforeAll(() => {
    if (!REAL_JWT) {
      throw new Error('[BLOCKED] REAL_JWT 환경변수가 없음. 실 JWT 없이 실 gateway 캡처 불가.')
    }
  })

  // ──────────────────────────────────────────────────────────
  // 캡처 1: "본사" 타이핑 → HQ-001 후보 드롭다운 노출
  //
  // 실 gateway GET /api/v1/inventory/warehouses 에서 창고 목록 로드.
  // WarehouseAutocomplete input 에 "본사" 타이핑 → listbox[role=listbox] 노출.
  // HQ-001 옵션이 candidates 에 포함됨을 실 화면으로 증명.
  // ──────────────────────────────────────────────────────────
  test('캡처1: "본사" 타이핑 → HQ-001 자동완성 후보 드롭다운 노출', async ({ page }) => {
    await installRealJwtStub(page, REAL_JWT)

    // 실 gateway 네트워크 요청 모니터링
    const warehouseApiHit: string[] = []
    page.on('response', async (res) => {
      if (res.url().includes('warehouses')) {
        warehouseApiHit.push(`${res.status()} GET ${res.url()}`)
        console.log(`[API HIT] warehouses: ${res.status()} ${res.url()}`)
      }
    })

    await page.goto(detailUrl(DRAFT_ORDER_UUID), { waitUntil: 'domcontentloaded' })
    await expect(page.getByText('주문서 상세')).toBeVisible({ timeout: 20_000 })
    await page.waitForTimeout(1500)

    // 출고전표 전환 버튼 클릭
    const convertBtn = page.getByTestId('partner-order-convert-open')
    await expect(convertBtn).toBeVisible({ timeout: 10_000 })
    await convertBtn.click()

    const modalBody = page.getByTestId('partner-order-convert-modal-body')
    await expect(modalBody).toBeVisible({ timeout: 10_000 })

    // WarehouseAutocomplete input — role=combobox
    const warehouseDiv = page.getByTestId('partner-order-convert-warehouse')
    await expect(warehouseDiv).toBeVisible({ timeout: 8_000 })
    const autocompleteInput = warehouseDiv.locator('input[role="combobox"]')
    await expect(autocompleteInput).toBeVisible({ timeout: 5_000 })

    // "본사" 타이핑
    await autocompleteInput.click()
    await autocompleteInput.fill('본사')
    // listbox 노출 대기 (aria-expanded=true)
    await expect(autocompleteInput).toHaveAttribute('aria-expanded', 'true', { timeout: 5_000 })

    const listbox = warehouseDiv.locator('[role="listbox"]')
    await expect(listbox).toBeVisible({ timeout: 5_000 })

    // HQ-001 후보 확인 — optionCode 또는 optionName span
    const firstOption = listbox.locator('[role="option"]').first()
    await expect(firstOption).toBeVisible({ timeout: 3_000 })
    const optionText = await firstOption.innerText()
    console.log('[AUTOCOMPLETE FIRST OPTION]', optionText)

    // 캡처 1: 후보 드롭다운 노출 상태
    await capture(page, 'ui-01-autocomplete-suggestions.png')

    // HQ-001 포함 증명 (code 또는 name)
    const allOptions = await listbox.locator('[role="option"]').all()
    const allTexts = await Promise.all(allOptions.map((o) => o.innerText()))
    console.log('[ALL AUTOCOMPLETE OPTIONS]', allTexts)
    const hasHq001 = allTexts.some((t) => t.includes('HQ-001') || t.includes('본사'))
    console.log('[HQ-001 IN CANDIDATES]', hasHq001)

    console.log('[WAREHOUSE API HITS]', warehouseApiHit)
  })

  // ──────────────────────────────────────────────────────────
  // 캡처 2: HQ-001 클릭 선택 → 입력에 "HQ-001 · 본사창고" 표시 + 수량 + 제출 활성
  //
  // listbox 첫 번째 후보 클릭 → input value = "HQ-001 · 본사창고"
  // 수량 입력 후 제출 버튼 enabled 확인.
  // ──────────────────────────────────────────────────────────
  test('캡처2: HQ-001 후보 클릭 선택 → "HQ-001 · 본사창고" + 제출 활성', async ({ page }) => {
    await installRealJwtStub(page, REAL_JWT)

    const apiResponses: string[] = []
    page.on('response', async (res) => {
      if (res.url().includes('warehouses') || res.url().includes('partner-orders')) {
        apiResponses.push(`${res.status()} ${res.url()}`)
      }
    })

    await page.goto(detailUrl(DRAFT_ORDER_UUID), { waitUntil: 'domcontentloaded' })
    await expect(page.getByText('주문서 상세')).toBeVisible({ timeout: 20_000 })
    await page.waitForTimeout(1500)

    const convertBtn = page.getByTestId('partner-order-convert-open')
    await expect(convertBtn).toBeVisible({ timeout: 10_000 })
    await convertBtn.click()

    const modalBody = page.getByTestId('partner-order-convert-modal-body')
    await expect(modalBody).toBeVisible({ timeout: 10_000 })

    // 수량 입력
    const qtyInput = page.getByTestId('partner-order-convert-qty-0')
    await expect(qtyInput).toBeVisible({ timeout: 5_000 })
    await qtyInput.fill('1')

    // WarehouseAutocomplete — "HQ" 타이핑 → 후보 노출 → 첫 번째 클릭
    const warehouseDiv = page.getByTestId('partner-order-convert-warehouse')
    const autocompleteInput = warehouseDiv.locator('input[role="combobox"]')
    await expect(autocompleteInput).toBeVisible({ timeout: 5_000 })
    await autocompleteInput.click()
    await autocompleteInput.fill('HQ')
    await expect(autocompleteInput).toHaveAttribute('aria-expanded', 'true', { timeout: 5_000 })

    const listbox = warehouseDiv.locator('[role="listbox"]')
    await expect(listbox).toBeVisible({ timeout: 5_000 })
    const firstOption = listbox.locator('[role="option"]').first()
    await expect(firstOption).toBeVisible({ timeout: 3_000 })
    await firstOption.click()

    // 선택 후 input 에 "HQ-001 · 본사창고" 표시 확인
    await expect(autocompleteInput).toHaveAttribute('aria-expanded', 'false', { timeout: 3_000 })
    const inputValue = await autocompleteInput.inputValue()
    console.log('[SELECTED INPUT VALUE]', inputValue)
    // 표시값이 HQ-001 을 포함하거나 "본사" 를 포함해야 함
    const isCorrectSelection = inputValue.includes('HQ-001') || inputValue.includes('본사')
    console.log('[CORRECT SELECTION]', isCorrectSelection, inputValue)

    await page.waitForTimeout(500)

    // 캡처 2: 선택 완료 + 수량 입력 + 제출 활성
    await capture(page, 'ui-02-selected.png')

    // 제출 버튼 활성 확인
    const submitBtn = page.getByTestId('partner-order-convert-submit')
    const isEnabled = await submitBtn.isEnabled()
    console.log('[SUBMIT ENABLED]', isEnabled)

    console.log('[API RESPONSES]', apiResponses.join('\n'))
  })

  // ──────────────────────────────────────────────────────────
  // 캡처 3: 전환 성공 → 실 slipNo 토스트
  //
  // 실 POST /api/v1/partner-orders/{uuid}/convert-to-slip → HTTP 200 → slipNo.
  // 토스트에 실 slipNo 패턴(YYYY/MM/DD-N 또는 SL-YYYYMMDD-N) 확인.
  // ──────────────────────────────────────────────────────────
  test('캡처3: 전환 성공 → 실 slipNo 토스트', async ({ page }) => {
    await installRealJwtStub(page, REAL_JWT)

    const convertApiLog: string[] = []
    page.on('response', async (res) => {
      if (res.url().includes('convert-to-slip')) {
        try {
          const body = await res.text()
          convertApiLog.push(`STATUS=${res.status()} URL=${res.url()} BODY=${body.slice(0, 300)}`)
          console.log('[CONVERT API RESPONSE]', res.status(), body.slice(0, 300))
        } catch { /* ignore */ }
      }
    })

    await page.goto(detailUrl(DRAFT_ORDER_UUID), { waitUntil: 'domcontentloaded' })
    await expect(page.getByText('주문서 상세')).toBeVisible({ timeout: 20_000 })
    await page.waitForTimeout(1500)

    const convertBtn = page.getByTestId('partner-order-convert-open')
    await expect(convertBtn).toBeVisible({ timeout: 10_000 })
    await convertBtn.click()

    const modalBody = page.getByTestId('partner-order-convert-modal-body')
    await expect(modalBody).toBeVisible({ timeout: 10_000 })

    // 수량 입력
    const qtyInput = page.getByTestId('partner-order-convert-qty-0')
    await expect(qtyInput).toBeVisible({ timeout: 5_000 })
    await qtyInput.fill('1')

    // 창고 자동완성 선택 — "HQ" → 첫 번째 후보 클릭
    const warehouseDiv = page.getByTestId('partner-order-convert-warehouse')
    const autocompleteInput = warehouseDiv.locator('input[role="combobox"]')
    await expect(autocompleteInput).toBeVisible({ timeout: 5_000 })
    await autocompleteInput.click()
    await autocompleteInput.fill('HQ')
    await expect(autocompleteInput).toHaveAttribute('aria-expanded', 'true', { timeout: 5_000 })
    const listbox = warehouseDiv.locator('[role="listbox"]')
    await expect(listbox).toBeVisible({ timeout: 5_000 })
    await listbox.locator('[role="option"]').first().click()
    await expect(autocompleteInput).toHaveAttribute('aria-expanded', 'false', { timeout: 3_000 })

    await page.waitForTimeout(300)

    // 제출
    const submitBtn = page.getByTestId('partner-order-convert-submit')
    await expect(submitBtn).toBeEnabled({ timeout: 3_000 })
    await submitBtn.click()

    // 성공 토스트 대기 (실 API 응답 시간 고려 15초)
    const toast = page.getByTestId('partner-order-convert-toast')
    await expect(toast).toBeVisible({ timeout: 15_000 })

    // 캡처 3: 성공 토스트
    await capture(page, 'ui-03-convert-success.png')

    const toastText = await toast.innerText()
    console.log('[TOAST TEXT]', toastText)

    // slipNo 패턴 확인 (실 슬립번호: YYYY/MM/DD-N 또는 SL-YYYYMMDD-N 등)
    const hasSlipNo = /\d{4}[\/\-]\d{2}[\/\-]\d{2}[-]\d+|SL-\d{8}-\d+/.test(toastText)
    console.log('[SLIP NO IN TOAST]', hasSlipNo, toastText)

    console.log('[CONVERT API LOG]', convertApiLog.join('\n'))
  })
})
